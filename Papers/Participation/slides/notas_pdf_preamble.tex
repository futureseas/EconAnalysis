\usepackage{xcolor}
\definecolor{udecblue}{HTML}{113B63}   % headings; prints near-black
\definecolor{accentgold}{HTML}{A8690A} % darker step of the deck's gold
\definecolor{warncol}{HTML}{8F2A2A}
\definecolor{speechbg}{HTML}{F2F4F7}
\definecolor{rulecol}{HTML}{C3CAD2}

\usepackage{pifont}
\usepackage{etoolbox}
\usepackage{fancyhdr}
\usepackage{tcolorbox}
\tcbuselibrary{skins, breakable}

% ---- the spoken script -------------------------------------------------
% Markdown blockquotes are exactly the lines to be said out loud, so they
% get the one strong visual signal in the document: a blue spine. breakable
% so a long script can run across a page turn.
\newtcolorbox{speechbox}{%
  enhanced, breakable, frame hidden,
  colback=speechbg, coltext=black!88,
  borderline west={2.4pt}{0pt}{udecblue},
  arc=1.5pt, boxsep=0pt, left=11pt, right=9pt, top=8pt, bottom=8pt,
  width=\linewidth, before skip=7pt, after skip=9pt,
}
% Each paragraph inside a box is a separate beat of the script, so they get
% more air than body paragraphs -- that gap is where you breathe.
\renewenvironment{quote}%
  {\begin{speechbox}\setlength{\parskip}{9pt plus 1pt}}%
  {\end{speechbox}}

% ---- section (= one slide) --------------------------------------------
% Sections stay NUMBERED internally -- \@ssect never sets marks, so an
% unnumbered section would leave \leftmark empty and kill the running head.
% The number is suppressed at print time instead.
\makeatletter
\renewcommand{\@seccntformat}[1]{}
\makeatother

\newcommand{\slideruleabove}{%
  \par\penalty-200\vspace{9pt}%
  \noindent{\color{rulecol}\rule{\linewidth}{0.6pt}}\par\vspace{-11pt}}
\pretocmd{\section}{\slideruleabove}{}{\PackageWarning{notas}{section hook failed}}

\makeatletter
\renewcommand\section{\@startsection{section}{1}{\z@}%
  {-1.6ex \@plus -0.6ex \@minus -.2ex}%
  {1.0ex \@plus .2ex}%
  {\normalfont\sffamily\bfseries\large\color{udecblue}}}
\renewcommand\subsection{\@startsection{subsection}{2}{\z@}%
  {-1.4ex \@plus -0.5ex \@minus -.2ex}%
  {0.7ex \@plus .2ex}%
  {\normalfont\sffamily\bfseries\normalsize\color{black!75}}}
\makeatother

% \pretocmd above must be re-applied: \section was redefined after it.
\pretocmd{\section}{\slideruleabove}{}{\PackageWarning{notas}{section hook failed}}

% ---- running head: which slide am I on ---------------------------------
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\headrule}{{\color{rulecol}%
  \hrule height \headrulewidth width \headwidth}}
\fancyhead[L]{\footnotesize\sffamily\color{black!45}IIFET 2026 \textbullet\ notas de charla}
\fancyhead[R]{\footnotesize\sffamily\color{udecblue}\nouppercase{\leftmark}}
\fancyfoot[C]{\footnotesize\sffamily\color{black!45}\thepage}
\setlength{\headheight}{14pt}

% ---- inline markers (substituted for emoji by make_notas_pdf.py) -------

% ---- compact title block -----------------------------------------------
% article's \maketitle burns a third of page 1 on white space; this is a
% masthead instead, so the running order starts above the fold.
\fancypagestyle{plain}{%
  \fancyhf{}\renewcommand{\headrulewidth}{0pt}%
  \fancyfoot[C]{\footnotesize\sffamily\color{black!45}\thepage}}
\makeatletter
\renewcommand{\maketitle}{%
  \thispagestyle{plain}%
  \noindent{\color{udecblue}\rule{\linewidth}{2pt}}\par\vspace{7pt}%
  {\sffamily\bfseries\LARGE\color{udecblue}\@title\par}%
  \vspace{5pt}%
  {\sffamily\footnotesize\color{black!55}\@date\par}%
  \vspace{5pt}%
  \noindent{\color{rulecol}\rule{\linewidth}{0.6pt}}\par
  \vspace{12pt}}
\makeatother
