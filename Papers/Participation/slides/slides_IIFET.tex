% ========================== 2. MOTIVATION ================================
\begin{frame}{Motivation --- the human margin of climate adaptation}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.55\textwidth}
\begin{itemize}
  \item Climate \textbf{reallocates} forage-species distributions --- the 2014--2016 heatwave shifted squid \textbf{northward}.
  \item To keep fishing, harvesters can \textbf{follow} species, \textbf{switch} species, \textbf{switch ports}, halt, or exit.
  \item Standard projections \textbf{fix behavior} --- missing the margin where fishers actually \emph{adapt}.
\end{itemize}
\end{column}
\begin{column}{0.45\textwidth}
\textbf{Question}\\[2pt]
How did \textbf{availability, prices, costs, weather} shape \textbf{daily} decisions to
\alert{participate}, \alert{target}, and \alert{land} --- over 2013--2017, spanning the
\textbf{2015 sardine closure} and the \textbf{2014--2016 heatwave}?
\end{column}
\end{columns}
\takeaway{We measure \textbf{demonstrated flexibility} across the joint choice
\emph{participate $\times$ target $\times$ land}.}
\end{frame}

% ========================== 3. CONTRIBUTION / SDM =======================
\begin{frame}{The key input: availability from \textbf{SDMs}, not past catch}
{\small We estimate a nested-logit RUM of the joint daily choice (participate/target/land),
by fleet segment. Its central covariate is \textbf{expected species availability} --- and \emph{how} we
measure it is the contribution.}
\vskip5pt
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{Past catch} {\scriptsize(the usual proxy)}
\begin{itemize}\small
  \item[\textcolor{red!70}{\ding{55}}] \textbf{Endogenous} --- reflects where vessels already \emph{chose} to fish.
  \item[\textcolor{red!70}{\ding{55}}] \textbf{Selection bias} --- only observed at fished sites (Chen et al., 2023).
  \item[\textcolor{red!70}{\ding{55}}] \textbf{Gaps} --- undefined on days/ports with no fishing.
\end{itemize}
\end{column}
\begin{column}{0.52\textwidth}
\textbf{SDM habitat index} {\scriptsize(ours)}
\begin{itemize}\small
  \item[\textcolor{udecgold}{\ding{51}}] \textbf{Exogenous} environmental input to the choice.
  \item[\textcolor{udecgold}{\ding{51}}] \textbf{No gaps} --- an availability value for \emph{every} port--day.
  \item[\textcolor{udecgold}{\ding{51}}] \textbf{Time-varying} --- captures environmentally-driven interannual shifts.
\end{itemize}
\end{column}
\end{columns}
\takeaway{An \textbf{exogenous}, gap-free, climate-responsive proxy $\Rightarrow$ cleaner identification
\emph{and} climate-ready counterfactuals.}
\end{frame}

% ========================== 5. MODEL ====================================
\begin{frame}{The model --- nested logit over \emph{participate $\times$ target $\times$ port}}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.54\textwidth}
\small
Utility of alternative $j=(p,s,l)$ for vessel $i$ on day $t$:
\[ U_{itj}= \delta_j + \beta_{1,s}\,\mathrm{Avail}_{tj} + \beta_{2,g}\,\mathrm{Price}_{tj} + \ldots + \varepsilon_{itj} \]
\begin{itemize}\footnotesize
  \item \textbf{Availability} = daily \textbf{SDM} index (species radius); CPUE index otherwise.
  \item Also: price, distances (catch, center of gravity), wind, closures, unemployment.
  \item \textbf{State dependence} (chosen in last 30 days) --- proxy for processor ties / port capacity.
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\textbf{Nesting (by segment)}
\begin{itemize}\footnotesize
  \item Correlation within \textbf{participation}, then within \textbf{species}\ldots
  \item \ldots except \textbf{forage-diverse}: nests by \textbf{port}.
  \item Crab/salmon split off (costly gear switch).
\end{itemize}
\vskip3pt
{\footnotesize Apollo (R). Dominant species $=$ target (\textbf{93.6\%} of trip revenue); $\sim$60/40 no-participation/trips.}
\end{column}
\end{columns}
\takeaway{Availability enters at \textbf{daily} frequency and is \textbf{exogenous} to the vessel's choice.}
\end{frame}

% ========================== 8. SUBSTITUTION STRUCTURE ====================
\begin{frame}{Two adaptation modes --- ``on the move'' vs.\ ``in place''}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.5\textwidth}
\textbf{Adapt \emph{on the move}} {\scriptsize(species nests)}
\begin{itemize}\small
  \item Low dissimilarity $\lambda$ $\Rightarrow$ strong within-species correlation across \textbf{ports}.
  \item Anchovy $\lambda=0.49$ (squid spec.); sardine $\lambda=0.40$ (roving, PNW).
  \item $\Rightarrow$ \textbf{follow the species across ports}.
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Adapt \emph{in place}} {\scriptsize(port nests)}
\begin{itemize}\small
  \item Forage-diverse nests by \textbf{port}; Santa Barbara $\lambda=0.49$ vs.\ LA/Monterey $\lambda=0.73$.
  \item Substitution across \textbf{species within a port}.
  \item $\Rightarrow$ \textbf{switch species, stay put} (by relative price).
\end{itemize}
\end{column}
\end{columns}
\takeaway{The adaptation \emph{mode} is a structural property of the fleet
(sensu Samhouri et al., 2024).}
\end{frame}

% ========================== 9. ELASTICITIES =============================
\begin{frame}{Elasticities --- price drives \emph{entry}, availability drives \emph{targeting}}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.5\textwidth}
\textbf{Extensive margin} {\scriptsize(participate)}
\begin{itemize}\small
  \item Price dominates for mobile fleets: squid \textbf{1.71} (roving); anchovy \textbf{0.43} (PNW).
  \item Non-CPS availability matters where crab/salmon are relevant: crab \textbf{0.24}, sockeye \textbf{0.19} (PNW).
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Intensive margin} {\scriptsize(conditional target)}
\begin{itemize}\small
  \item Availability often $>$ price: anchovy \textbf{2.55}, crab \textbf{2.12} (roving); crab \textbf{3.85}, sockeye \textbf{3.71} (PNW).
  \item Several \textbf{$>1$} $\Rightarrow$ more-than-proportional responses.
  \item Exception: forage-diverse is \textbf{price}-driven.
\end{itemize}
\end{column}
\end{columns}
\takeaway{\textbf{Prices} move participation; \textbf{availability} moves which species, once at sea.}
\end{frame}

% ========================== 11. TAKEAWAYS ==============================
\begin{frame}{Takeaways}
\begin{enumerate}
  \item Harvesters are \textbf{flexible}: participation, target, and landing respond to
        \textbf{availability, prices, costs, weather} --- \textbf{even controlling for strong state
        dependence}, so it is not just habit.
  \item \textbf{Two adaptation modes} --- ``on the move'' (substitute across \emph{ports}) vs.\
        ``in place'' (switch \emph{species} within a port).
  \item \textbf{SDM-based} availability is exogenous and gap-free; OOS pseudo-$\rho^2$ \textbf{0.22--0.44}
        $\Rightarrow$ a reliable, climate-ready policy tool.
\end{enumerate}
\takeaway{Adaptive capacity is a property of the \textbf{bundle} (portfolio $\times$ ports $\times$
institutions $\times$ market), not of fishers alone.}
\end{frame}

\begin{frame}{Backup \textbullet\ Variable construction}
\small
\begin{itemize}
  \item \textbf{Availability:} binomial GAM SDMs (Muhling et al., 2019, 2020); mean presence within a
        species radius (60 km sardine/anchovy/mackerel, 90 km squid). CPUE $z$-score $\to$ sigmoid for non-SDM species.
  \item \textbf{Price:} predicted from $\ln(\text{price})=\alpha+T_y+\gamma_m+\gamma_p$ (year trend, month \& port FE);
        robustness with 30-day moving average.
  \item \textbf{Costs:} Haversine distance port$\to$catch area and port$\to$center of gravity.
  \item \textbf{Weather:} max ERA5 surface wind within 220 km. \textbf{Opportunity cost:} state unemployment.
\end{itemize}
\end{frame}

\begin{frame}{Backup \textbullet\ Limitations}
\small
\begin{itemize}
  \item \textbf{Aggregation bias:} port-area resolution may mask finer spatial heterogeneity (D\'epalle et al., 2021).
  \item \textbf{Choice-set truncation:} rare alternatives folded into the outside option ($\ge$0.5\% filter); IIA assumed for omitted alternatives.
  \item \textbf{Price endogeneity:} predicted / MA prices reduce but do not eliminate simultaneity $\Rightarrow$
        interpret price coefficients as \textbf{reduced-form} behavioral responses.
  \item \textbf{SDM proxy:} habitat suitability $\ne$ catch rate exactly $\Rightarrow$ availability effect may be attenuated.
\end{itemize}
\takeaway{Main qualitative result --- \textbf{availability drives targeting} --- is robust across price specifications.}
\end{frame}
