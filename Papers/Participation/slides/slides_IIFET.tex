% ========================== 2. MOTIVATION ==============================
% ~50 s. Land the question; do not linger.
\begin{frame}{Motivation --- spatial trade-offs run through the \textbf{species portfolio}}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.53\textwidth}
\begin{itemize}
  \item Fishable area shrinking (\textbf{wind, conservation, mining}); species moving
        (\textbf{climate}).
  \item What matters is how fleets \textbf{redistribute effort} --- follow, switch species,
        switch ports, halt, exit.
  \item Location choice usually models \textbf{one fishery at a time} $\Rightarrow$ the
        \textbf{species portfolio} drops out.
\end{itemize}
\end{column}
\hfill
\begin{column}{0.43\textwidth}
\textbf{Question}\\[2pt]
How do \textbf{availability, prices, costs, weather} shape \textbf{daily} choices to
\alert{participate}, \alert{target}, \alert{land}?\\[4pt]
2013--2017: the \textbf{2015 sardine closure} and the \textbf{2014--2016 heatwave}.
\end{column}
\end{columns}
\takeaway{Where the boats end up is produced by \textbf{which species they can switch to}
--- so we model \emph{participate $\times$ target $\times$ land} \textbf{jointly}.}
\end{frame}

% ========================== 4. WHY IT MATTERS ==========================
% ~55 s. The three-against-three contrast. Read only the bold words.
\begin{frame}{Why that swap matters --- past catch vs.\ SDM}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.46\textwidth}
\textbf{Past catch} {\scriptsize(the usual proxy)}
\begin{itemize}\small
  \item[\textcolor{red!70}{\ding{55}}] \textbf{Endogenous} --- where vessels already \emph{chose} to fish.
  \item[\textcolor{red!70}{\ding{55}}] \textbf{Selection bias} --- only at fished sites (Chen et al., 2023).
  \item[\textcolor{red!70}{\ding{55}}] \textbf{Gaps} --- undefined when nobody fishes.
\end{itemize}
\end{column}
\hfill
\begin{column}{0.50\textwidth}
\textbf{SDM habitat index} {\scriptsize(ours)}
\begin{itemize}\small
  \item[\textcolor{udecgold}{\ding{51}}] \textbf{Exogenous} --- an environmental input.
  \item[\textcolor{udecgold}{\ding{51}}] \textbf{No gaps} --- a value for \emph{every} port--day.
  \item[\textcolor{udecgold}{\ding{51}}] \textbf{Counterfactual-ready} --- move the field, re-predict.
\end{itemize}
\end{column}
\end{columns}
\takeaway{The third point buys us the \textbf{scenario} --- you cannot shift a
``past catch'' field.}
\end{frame}

% ========================== 6. MODEL ===================================
% ~50 s. [CUT-able to 25 s] Do not derive anything. One line of utility,
% one line on nesting, move on.
\begin{frame}{The model --- nested logit over \emph{participate $\times$ target $\times$ port}}
\small
Utility of alternative $j=(p,s,l)$ for vessel $i$ on day $t$:
\[
  U_{itj}= \delta_j + \alert{\beta_{1,s}\,\mathrm{Avail}_{tj}}
         + \beta_{2,g}\,\mathrm{Price}_{tj} + \ldots + \varepsilon_{itj}
\]
\vskip-2pt
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\begin{itemize}\footnotesize
  \item \alert{$\mathrm{Avail}$} $=$ the daily \textbf{SDM} index from the previous slide.
  \item Also: price, distances, wind, closures, unemployment.
  \item \textbf{State dependence} (last 30 days) --- processor ties / port capacity.
\end{itemize}
\end{column}
\hfill
\begin{column}{0.46\textwidth}
\begin{itemize}\footnotesize
  \item Nests: \textbf{participation}, then \textbf{species} --- except forage-diverse,
        which nests by \textbf{port}.
  \item Crab/salmon split off (costly gear switch).
  \item Apollo (R); dominant species $=$ target (\textbf{93.6\%} of revenue).
\end{itemize}
\end{column}
\end{columns}
\takeaway{Availability enters at \textbf{daily} frequency and is \textbf{exogenous} to the vessel's choice.}
\end{frame}

% ========================== 8. SUBSTITUTION STRUCTURE ===================
% ~45 s. [CUT] Drop this frame first if the session is running late --
% the takeaways slide still states both modes.
\begin{frame}{Two adaptation modes --- ``on the move'' vs.\ ``in place''}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.47\textwidth}
\textbf{Adapt \emph{on the move}} {\scriptsize(species nests)}
\begin{itemize}\small
  \item Low $\lambda$ $\Rightarrow$ strong correlation across \textbf{ports}, same species.
  \item Anchovy $\lambda=0.49$; sardine $\lambda=0.40$.
  \item $\Rightarrow$ \textbf{follow the species across ports}.
\end{itemize}
\end{column}
\hfill
\begin{column}{0.49\textwidth}
\textbf{Adapt \emph{in place}} {\scriptsize(port nests)}
\begin{itemize}\small
  \item Forage-diverse nests by \textbf{port}: Santa Barbara $\lambda=0.49$ vs.\ LA/Monterey $0.73$.
  \item Substitution across \textbf{species within a port}.
  \item $\Rightarrow$ \textbf{switch species, stay put}.
\end{itemize}
\end{column}
\end{columns}
\takeaway{The mode is a \textbf{structural property of the fleet} (sensu Samhouri et al., 2024)
--- and it predicts who absorbs a \textbf{closed area}: ``on the move'' fleets re-route,
``in place'' fleets cannot.}
\end{frame}

% ========================== 10. TAKEAWAYS ==============================
% ~55 s. If time is gone, this frame alone can close the talk.
\begin{frame}{Takeaways}
\begin{enumerate}
  \item \textbf{Oceanographic SDMs work as economic regressors.} An exogenous, gap-free
        daily availability index replaces the endogenous past-catch proxy --- and
        out-of-sample pseudo-$\rho^2$ is \textbf{0.22--0.44}.
  \item \textbf{Harvesters are flexible.} Participation, target and landing all respond to
        availability, prices, costs and weather --- \emph{even} controlling for strong
        state dependence.
  \item \textbf{The spatial answer runs through the portfolio.} Two adaptation modes ---
        ``on the move'' (across \emph{ports}) vs.\ ``in place'' (switch \emph{species}
        within a port) --- and Los Angeles shows why it matters: squid targeting there
        \emph{rises} ($+1.5$\,pp), yet the port \textbf{loses} effort ($-0.6$) because
        mackerel falls more.
\end{enumerate}
\takeaway{Availability is an \textbf{ocean-model output}, so the same machinery runs forward
under \textbf{climate projections} --- and takes a \textbf{closed area} as just another change
in the spatial choice set.}
\end{frame}

\begin{frame}{Backup \textbullet\ Variable construction}
\small
\begin{itemize}
  \item \textbf{Availability:} binomial GAM SDMs (Muhling et al., 2019, 2020); mean presence within a
        species radius (60 km sardine/anchovy/mackerel, 90 km squid). CPUE $z$-score $\to$ sigmoid for non-SDM species.
  \item \textbf{Price:} predicted from $\ln(\text{price})=\alpha+T_y+\gamma_m+\gamma_p$ (year trend, month \& port FE);
        robustness with 30-day moving average.
  \item \textbf{Costs:} Haversine distance port$\to$catch area and port$\to$center of gravity.
  \item \textbf{Weather:} max ERA5 surface wind within 220 km. \textbf{Opportunity cost:} state unemployment.
\end{itemize}
\end{frame}

\begin{frame}{Backup \textbullet\ Limitations}
\small
\begin{itemize}
  \item \textbf{Aggregation bias:} port-area resolution may mask finer spatial heterogeneity (D\'epalle et al., 2021).
  \item \textbf{Choice-set truncation:} rare alternatives folded into the outside option ($\ge$0.5\% filter); IIA assumed for omitted alternatives.
  \item \textbf{Price endogeneity:} predicted / MA prices reduce but do not eliminate simultaneity $\Rightarrow$
        interpret price coefficients as \textbf{reduced-form} behavioral responses.
  \item \textbf{SDM proxy:} habitat suitability $\ne$ catch rate exactly $\Rightarrow$ availability effect may be attenuated.
\end{itemize}
\takeaway{Main qualitative result --- \textbf{availability drives targeting} --- is robust across price specifications.}
\end{frame}
